# Remove Unused Docker Images, Containers, and Volumes
docker system prune -a --volumes --force

# Remove dangling/untagged Docker images only
docker image prune -a --filter "dangling=true" --force

# Remove volumes not used by any containers
docker volume ls -qf dangling=true | while read volume; do
  # Get the creation time of the volume (approximate last use)
  created=$(docker volume inspect "$volume" --format '{{.CreatedAt}}')
  # Convert the creation date to epoch time
  created_epoch=$(date -d "$created" +%s)
  # Get the current date in epoch time
  now=$(date +%s)
  # Calculate the difference in days
  diff_days=$(( (now - created_epoch) / 86400 ))

  # Remove the volume if it was created more than 90 days ago
  if [ $diff_days -gt 90 ]; then
    echo "Removing volume: $volume (unused for more than 3 months)"
    docker volume rm "$volume"
  fi
done
